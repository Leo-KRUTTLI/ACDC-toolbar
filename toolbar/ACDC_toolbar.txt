<startupAction>

url = "https://raw.githubusercontent.com/Leo-KRUTTLI/MCZ-marco/refs/heads/main/MACRO%20MCZ%20update%20version.ijm";

dirMacros = getDirectory("macros") + "MACRO" + File.separator;

if (!File.exists(dirMacros)) {
    File.makeDirectory(dirMacros);
}

fileName = "MACRO_MCZ.ijm";
fullPath = dirMacros + fileName;

// 2. RECUPERATION DU CODE (LIGNE INDISPENSABLE)
code = File.openUrlAsString(url);

// 3. VERIFICATION ET ENREGISTREMENT
if (startsWith(code, "<Error") || code == "") {
    // ÉCHEC DU TÉLÉCHARGEMENT (Hors-ligne)
    if (File.exists(fullPath)) {
        print("Mode Hors-ligne : Chargement de la version locale.");
    } else {
        showMessage("Erreur", "Première installation impossible sans connexion internet.");
    }
} else {
    // SUCCÈS : Enregistrement sur le disque
    File.saveString(code, fullPath);
    showStatus("Macro MCZ mise à jour avec succès !");
}

// ==========================================
// 2. INSTALLATION OPTIONNELLE DES CHRIS LUTs
// ==========================================
dirLuts = getDirectory("luts");
flagFile = getDirectory("macros") + ".chris_luts_asked.txt"; 

if (!File.exists(flagFile)) {
    installLUTs = getBoolean("Voulez-vous télécharger et installer les Chris LUTs ?");

    if (installLUTs) {
        lutURLs = newArray(
            "https://raw.githubusercontent.com/cleterrier/ChrisLUTs/refs/heads/master/BOP%20blue.lut",
            "https://raw.githubusercontent.com/cleterrier/ChrisLUTs/refs/heads/master/BOP%20orange.lut",
            "https://raw.githubusercontent.com/cleterrier/ChrisLUTs/refs/heads/master/BOP%20purple.lut",
            "https://raw.githubusercontent.com/cleterrier/ChrisLUTs/refs/heads/master/I%20Blue.lut",
            "https://raw.githubusercontent.com/cleterrier/ChrisLUTs/refs/heads/master/I%20Bordeaux.lut",
            "https://raw.githubusercontent.com/cleterrier/ChrisLUTs/refs/heads/master/I%20Cyan.lut",
            "https://raw.githubusercontent.com/cleterrier/ChrisLUTs/refs/heads/master/I%20Forest.lut",
            "https://raw.githubusercontent.com/cleterrier/ChrisLUTs/refs/heads/master/I%20Green.lut",
            "https://raw.githubusercontent.com/cleterrier/ChrisLUTs/refs/heads/master/I%20Magenta.lut",
            "https://raw.githubusercontent.com/cleterrier/ChrisLUTs/refs/heads/master/I%20Purple.lut",
            "https://raw.githubusercontent.com/cleterrier/ChrisLUTs/refs/heads/master/I%20Red.lut",
            "https://raw.githubusercontent.com/cleterrier/ChrisLUTs/refs/heads/master/I%20Yellow.lut"
        );
        
        lutNames = newArray(
            "BOP blue.lut", "BOP orange.lut", "BOP purple.lut", 
            "I blue.lut", "I bordeaux.lut", "I cyan.lut", 
            "I forest.lut", "I Green.lut", "I Magenta.lut", 
            "I Purple.lut", "I Red.lut", "I Yellow.lut"
        );

        // Détection de l'OS compatible avec le langage Macro
        os = getInfo("os.name");

        for (i = 0; i < lutURLs.length; i++) {
            destPath = dirLuts + lutNames[i];
            
            if (startsWith(os, "Mac") || startsWith(os, "Linux")) {
                exec("curl", "-s", "-L", "-o", destPath, lutURLs[i]);
            } else {
                exec("powershell", "-Command", "Invoke-WebRequest -Uri '" + lutURLs[i] + "' -OutFile '" + destPath + "'");
            }
        }
        
        print("Les Chris LUTs ont été installées dans : " + dirLuts);
        showMessage("Installation terminée", "Les Chris LUTs sont installées ! Redémarrez Fiji pour qu'elles apparaissent dans Image > Lookup Tables.");
    }

    // Fichier témoin
    File.saveString("Question posée le : " + getTime(), flagFile);
}
</startupAction>

<button>
label=on/off
icon=noicon
arg=<macro>

Stack.getDimensions(width, height, channels, slices, frames);

Stack.getPosition(channel, slice, frame);

active = "";
Stack.getActiveChannels(active);

// sécurité si vide
if (active == "")
    active = "11110000";

// construire nouvelle chaîne
newActive = "";

for (i = 0; i < lengthOf(active); i++) {

    if (i == channel - 1) {

        // toggle 0 ↔ 1
        if (substring(active, i, i+1) == "1")
            newActive += "0";
        else
            newActive += "1";

    } else {
        newActive += substring(active, i, i+1);
    }
}

Stack.setActiveChannels(newActive);
</macro>
