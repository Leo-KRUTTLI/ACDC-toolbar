url = "https://raw.githubusercontent.com/Leo-KRUTTLI/MCZ-marco/refs/heads/main/MACRO%20MCZ%20update%20version.ijm";

dirMacros = getDirectory("macros"); // Ne pas ajouter File.separator ici
fileName = "MACRO_MCZ.ijm";
fullPath = dirMacros + fileName;

// 2. RECUPERATION DU CODE (LIGNE INDISPENSABLE)
code = File.openUrlAsString(url);

// 3. VERIFICATION ET ENREGISTREMENT
if (startsWith(code, "<Error") || code == "") {
    // ÉCHEC DU TÉLÉCHARGEMENT (Hors-ligne)
    if (File.exists(fullPath)) {
        print("Mode Hors-ligne : Chargement de la version locale.");
    } else {
        showMessage("Erreur", "Première installation impossible sans connexion internet.");
    }
} else {
    // SUCCÈS : Enregistrement sur le disque
    File.saveString(code, fullPath);
    showStatus("Macro MCZ mise à jour avec succès !");
}


</startupAction>

<button>
label=on/off
icon=noicon
arg=<macro>

Stack.getDimensions(width, height, channels, slices, frames);

Stack.getPosition(channel, slice, frame);

active = "";
Stack.getActiveChannels(active);

// sécurité si vide
if (active == "")
    active = "11110000";

// construire nouvelle chaîne
newActive = "";

for (i = 0; i < lengthOf(active); i++) {

    if (i == channel - 1) {

        // toggle 0 ↔ 1
        if (substring(active, i, i+1) == "1")
            newActive += "0";
        else
            newActive += "1";

    } else {
        newActive += substring(active, i, i+1);
    }
}

Stack.setActiveChannels(newActive);
</macro>
